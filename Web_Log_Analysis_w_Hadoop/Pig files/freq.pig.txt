-— Loading the data

queries = LOAD 'logs/web_log.csv' USING PigStorage(‘,')
		AS (user_id: chararray, search_query: chararray, browser: chararray, os: chararray, referrer: chararray);

-— Getting the distinct search counts

searches = FOREACH queries GENERATE search_query;
grouped_searches = GROUP searches BY search_query;
search_counts = FOREACH grouped_searches GENERATE group as search_query, COUNT(searches) AS count;
sorted_searches = ORDER search_counts BY count DESC;

—- Storing the results

STORE sorted_searches INTO 'logs/output' USING PigStorage(',');
